   230→function GlobalSearch() {
   231→  const [query, setQuery] = useState("");
   232→  const [open, setOpen] = useState(false);
   233→  const setView = useNavStore((s) => s.setView);
   234→
   235→  const { data: searchResults } = useQuery({
   236→    queryKey: ["global-search", query],
   237→    queryFn: async () => {
   238→      if (!query || query.length < 2) return null;
   239→      // Search across rough, polished, requirements, cases
   240→      try {
   241→        const [roughRes, polishedRes, reqRes] = await Promise.all([
   242→          apiFetch<{ rows: Array<{ id: string; fantasyRoughId: string; stoneName: string; kapan: string }> }>(`/api/planning/rough?q=${encodeURIComponent(query)}`).catch(() => null),
   243→          apiFetch<{ rows: Array<{ id: string; fantasyLotId: string; shape: string; weight: number }> }>(`/api/fantasy/polished?q=${encodeURIComponent(query)}`).catch(() => null),
   244→          apiFetch<{ data: Array<{ id: string; requirementCode: string; customerName: string | null }> }>(`/api/requirements?q=${encodeURIComponent(query)}`).catch(() => null),
   245→        ]);
   246→        return { rough: roughRes?.rows?.slice(0, 5) ?? [], polished: polishedRes?.rows?.slice(0, 5) ?? [], requirements: reqRes?.data?.slice(0, 5) ?? [] };
   247→      } catch {
   248→        return null;
   249→      }
   250→    },
   251→    enabled: query.length >= 2,
   252→    staleTime: 10_000,
   253→  });
   254→
   255→  return (
   256→    <div className="relative w-full max-w-md">
   257→      <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   258→      <Input
   259→        value={query}
   260→        onChange={(e) => { setQuery(e.target.value); setOpen(true); }}
   261→        onFocus={() => setOpen(true)}
   262→        onBlur={() => setTimeout(() => setOpen(false), 200)}
   263→        placeholder="Search Lot ID, Rough ID, Kapan, Requirement..."
   264→        className="h-8 pl-8 text-xs bg-muted/50 border-border/60"
   265→      />
   266→      {open && searchResults && (searchResults.rough.length > 0 || searchResults.polished.length > 0 || searchResults.requirements.length > 0) && (
   267→        <div className="absolute top-full mt-1 left-0 right-0 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   268→          {searchResults.rough.length > 0 && (
   269→            <div className="p-2">
   270→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Rough Stones</p>
   271→              {searchResults.rough.map((r) => (
   272→                <button key={r.id} onClick={() => { setView("fantasy-rough"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   273→                  <Gem className="h-3 w-3 text-muted-foreground" />
   274→                  <span className="font-medium">{r.fantasyRoughId}</span>
   275→                  <span className="text-muted-foreground truncate">{r.stoneName}</span>
   276→                </button>
   277→              ))}
   278→            </div>
   279→          )}
   280→          {searchResults.polished.length > 0 && (
   281→            <div className="p-2 border-t border-border">
   282→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Polished Lots</p>
   283→              {searchResults.polished.map((p) => (
   284→                <button key={p.id} onClick={() => { setView("fantasy-polished"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   285→                  <Diamond className="h-3 w-3 text-muted-foreground" />
   286→                  <span className="font-medium">{p.fantasyLotId}</span>
   287→                  <span className="text-muted-foreground">{p.shape} {p.weight}ct</span>
   288→                </button>
   289→              ))}
   290→            </div>
   291→          )}
   292→          {searchResults.requirements.length > 0 && (
   293→            <div className="p-2 border-t border-border">
   294→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Requirements</p>
   295→              {searchResults.requirements.map((r) => (
   296→                <button key={r.id} onClick={() => { setView("requirements-matrix"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   297→                  <ClipboardList className="h-3 w-3 text-muted-foreground" />
   298→                  <span className="font-medium">{r.requirementCode}</span>
   299→                  <span className="text-muted-foreground truncate">{r.customerName ?? ""}</span>
   300→                </button>
   301→              ))}
   302→            </div>
   303→          )}
   304→        </div>
   305→      )}
   306→    </div>
   307→  );
   308→}

   310→function ThemeToggle() {
   311→  const { theme, setTheme } = useTheme();
   312→  const [mounted, setMounted] = useState(false);
   313→  // eslint-disable-next-line react-hooks/set-state-in-effect

   315→  if (!mounted) return <div className="h-8 w-8" />;
   316→  return (
   317→    <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setTheme(theme === "dark" ? "light" : "dark")}>
   318→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   319→    </Button>
   320→  );
   321→}

   323→function NotificationsBell() {
   324→  const { data } = useQuery({
   325→    queryKey: ["notifications"],
   326→    queryFn: () => apiFetch<{ rows: Array<{ id: string; title: string; message: string; severity: string; read: boolean }> }>("/api/notifications"),
   327→    staleTime: 60_000,
   328→  });
   329→  const unread = data?.rows.filter((n) => !n.read).length ?? 0;
   330→  const [open, setOpen] = useState(false);
   331→  // Realtime notifications
   332→  const realtimeEvents = useRealtimeStore((s) => s.events);
   333→  const realtimeConnected = useRealtimeStore((s) => s.connected);
   334→  const realtimeUnread = useRealtimeStore((s) => s.unreadCount);
   335→  const clearRealtimeUnread = useRealtimeStore((s) => s.clearUnread);
   336→  const totalUnread = unread + realtimeUnread;
   337→
   338→  const handleToggle = () => {
   339→    const next = !open;
   340→    setOpen(next);
   341→    if (next && realtimeUnread > 0) clearRealtimeUnread();
   342→  };
   343→
   344→  const formatRelTime = (iso: string) => {
   345→    const diff = Date.now() - new Date(iso).getTime();
   346→    const sec = Math.floor(diff / 1000);
   347→    if (sec < 60) return `${sec}s ago`;
   348→    const min = Math.floor(sec / 60);
   349→    if (min < 60) return `${min}m ago`;
   350→    const hr = Math.floor(min / 60);
   351→    return `${hr}h ago`;
   352→  };
   353→
   354→  return (
   355→    <div className="relative">
   356→      <Button variant="ghost" size="icon" className="h-8 w-8 relative" onClick={handleToggle}>
   357→        <Bell className="h-4 w-4" />
   358→        {/* Live indicator when connected */}
   359→        {realtimeConnected && (
   360→          <span className="absolute top-1 right-1 h-2 w-2 rounded-full bg-emerald-500 ring-2 ring-card animate-pulse" title="Live — connected to realtime service" />
   361→        )}
   362→        {totalUnread > 0 && (
   363→          <span className="absolute -top-0.5 -right-0.5 h-3.5 min-w-3.5 px-1 rounded-full bg-rose-500 text-white text-[9px] font-bold flex items-center justify-center">
   364→            {totalUnread > 9 ? "9+" : totalUnread}
   365→          </span>
   366→        )}
   367→      </Button>
   368→      {open && (
   369→        <div className="absolute top-full mt-1 right-0 w-80 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   370→          <div className="px-3 py-2 border-b border-border bg-muted/50 flex items-center justify-between">
   371→            <p className="text-xs font-semibold">Notifications</p>
   372→            <span className={cn("flex items-center gap-1 text-[10px] font-medium px-1.5 py-0.5 rounded",
   373→              realtimeConnected ? "text-emerald-700 bg-emerald-100 dark:text-emerald-300 dark:bg-emerald-950/40" : "text-muted-foreground bg-muted")}>
   374→              <span className={cn("h-1.5 w-1.5 rounded-full", realtimeConnected ? "bg-emerald-500" : "bg-muted-foreground")} />
   375→              {realtimeConnected ? "Live" : "Offline"}
   376→            </span>
   377→          </div>
   378→          <div className="max-h-80 overflow-y-auto">
   379→            {/* Realtime events first */}
   380→            {realtimeEvents.length > 0 && (
   381→              <>
