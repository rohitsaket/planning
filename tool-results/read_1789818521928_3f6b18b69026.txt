   170→   170→function WarningsCell({ value }: { value: string | null }) {
   171→   171→  if (!value) return <span className="text-muted-foreground">—</span>;
   172→   172→  let parsed: string[] = [];
   173→   173→  try {
   174→   174→    const j = JSON.parse(value);
   175→   175→    if (Array.isArray(j)) parsed = j.map((s) => String(s));
   176→   176→    else if (typeof j === "string") parsed = [j];
   177→   177→    else parsed = [JSON.stringify(j)];
   178→   178→  } catch {
   179→   179→    parsed = [value];
   180→   180→  }
   181→   181→  if (parsed.length === 0) return <span className="text-muted-foreground">—</span>;
   182→   182→  return (
   183→   183→    <div className="flex flex-wrap gap-1">
   184→   184→      {parsed.slice(0, 3).map((w, i) => (
   185→   185→        <Badge key={i} variant="warning">{w}</Badge>
   186→   186→      ))}
   187→   187→      {parsed.length > 3 && (
   188→   188→        <Badge variant="neutral">+{parsed.length - 3}</Badge>
   189→   189→      )}
   190→   190→    </div>
   191→   191→  );
   192→   192→}
