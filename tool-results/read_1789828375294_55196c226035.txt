   149→
   150→function priorityVariant(
   151→  v: string | null
   152→): "default" | "critical" | "high" | "low" | "neutral" {
   153→  if (!v) return "default";
   154→  if (v === "CRITICAL") return "critical";
   155→  if (v === "HIGH") return "high";
   156→  if (v === "LOW") return "low";
   157→  if (v === "WATCH") return "neutral";
   158→  return "default";
   159→}
   160→
   161→function PriorityBadge({ value }: { value: string | null }) {
   162→  if (!value) return <span className="text-muted-foreground">—</span>;
   163→  return <Badge variant={priorityVariant(value)}>{value}</Badge>;
   164→}
   165→
   166→function fmtDate(iso: string | null): string {
   167→  if (!iso) return "—";
   168→  try {
   169→    return new Date(iso).toISOString().slice(0, 10);
   170→  } catch {
   171→    return "—";
   172→  }
   173→}
