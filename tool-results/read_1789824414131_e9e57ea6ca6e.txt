   148→
   149→function priorityVariant(
   150→  v: string | null
   151→): "default" | "critical" | "high" | "low" | "neutral" {
   152→  if (!v) return "default";
   153→  if (v === "CRITICAL") return "critical";
   154→  if (v === "HIGH") return "high";
   155→  if (v === "LOW") return "low";
   156→  if (v === "WATCH") return "neutral";
   157→  return "default";
   158→}
   159→
   160→function PriorityBadge({ value }: { value: string | null }) {
   161→  if (!value) return <span className="text-muted-foreground">—</span>;
   162→  return <Badge variant={priorityVariant(value)}>{value}</Badge>;
   163→}
   164→
   165→function fmtDate(iso: string | null): string {
   166→  if (!iso) return "—";
   167→  try {
   168→    return new Date(iso).toISOString().slice(0, 10);
   169→  } catch {
   170→    return "—";
   171→  }
   172→}
