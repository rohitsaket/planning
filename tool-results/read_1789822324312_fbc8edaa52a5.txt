   147→
   148→function priorityVariant(
   149→  v: string | null
   150→): "default" | "critical" | "high" | "low" | "neutral" {
   151→  if (!v) return "default";
   152→  if (v === "CRITICAL") return "critical";
   153→  if (v === "HIGH") return "high";
   154→  if (v === "LOW") return "low";
   155→  if (v === "WATCH") return "neutral";
   156→  return "default";
   157→}
   158→
   159→function PriorityBadge({ value }: { value: string | null }) {
   160→  if (!value) return <span className="text-muted-foreground">—</span>;
   161→  return <Badge variant={priorityVariant(value)}>{value}</Badge>;
   162→}
   163→
   164→function fmtDate(iso: string | null): string {
   165→  if (!iso) return "—";
   166→  try {
   167→    return new Date(iso).toISOString().slice(0, 10);
   168→  } catch {
   169→    return "—";
   170→  }
   171→}
