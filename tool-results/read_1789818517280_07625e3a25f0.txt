   130→
   131→function priorityVariant(
   132→  v: string | null
   133→): "default" | "critical" | "high" | "low" | "neutral" {
   134→  if (!v) return "default";
   135→  if (v === "CRITICAL") return "critical";
   136→  if (v === "HIGH") return "high";
   137→  if (v === "LOW") return "low";
   138→  if (v === "WATCH") return "neutral";
   139→  return "default";
   140→}
   141→
   142→function PriorityBadge({ value }: { value: string | null }) {
   143→  if (!value) return <span className="text-muted-foreground">—</span>;
   144→  return <Badge variant={priorityVariant(value)}>{value}</Badge>;
   145→}
   146→
   147→function fmtDate(iso: string | null): string {
   148→  if (!iso) return "—";
   149→  try {
   150→    return new Date(iso).toISOString().slice(0, 10);
   151→  } catch {
   152→    return "—";
   153→  }
   154→}

   599→              <div className="inline-flex items-center gap-2">
   600→                <div className="h-3 w-3 border-2 border-muted-foreground border-t-transparent rounded-full animate-spin" />
   601→                Loading requirement detail…
   602→              </div>
   603→            </div>
   604→          ) : (
   605→            <div className="flex flex-col gap-3">
   606→              {/* 4 numbers grid */}
   607→              <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   608→                <KpiCard
   609→                  label="1 · Physical Shortage"
   610→                  value={detail.fourNumbers.physicalShortage}
   611→                  unit="pcs"
   612→                  intent="critical"
   613→                  hint="MAX(0, RequiredQty − PlanningAvailableQty)"
   614→                />
   615→                <KpiCard
   616→                  label="2 · Pipeline-Adjusted"
   617→                  value={detail.fourNumbers.pipelineAdjusted}
   618→                  unit="pcs"
   619→                  intent="warning"
   620→                  hint="MAX(0, PhysicalShortage − WIP Coverage)"
   621→                />
   622→                <KpiCard
   623→                  label="3 · Planning-Adjusted"
   624→                  value={detail.fourNumbers.planningAdjusted}
   625→                  unit="pcs"
   626→                  intent="info"
   627→                  hint="MAX(0, PipelineAdjusted − ApprovedPlanCoverage)"
   628→                />
   629→                <KpiCard
   630→                  label="4 · Forecast Signal"
   631→                  value={detail.fourNumbers.forecastRequirement}
   632→                  unit="pcs"
   633→                  intent="default"

   635→                />
   636→              </div>
   637→
   638→              {/* Quantities breakdown */}
   639→              <div className="grid grid-cols-2 md:grid-cols-4 gap-2 text-[11px]">
   640→                <DataPair label="Required Qty" value={detail.requiredQty} />
   641→                <DataPair label="Physical Stock" value={detail.physicalStockQty} />
   642→                <DataPair label="Planning Available" value={detail.planningAvailableQty} />

   645→                <DataPair label="WIP Coverage" value={detail.wipCoverage} />
   646→                <DataPair label="Approved Plan Coverage" value={detail.approvedPlanCoverage} />
   647→                <DataPair label="Actual Coverage" value={detail.actualCoverage} />
   648→              </div>
   649→
   650→              {/* Context */}
   651→              <div className="grid grid-cols-2 md:grid-cols-3 gap-2 text-[11px] rounded-md border border-border bg-muted/30 p-2">
   652→                <DataPair label="Customer" value={detail.customerName ?? "—"} />
   653→                <DataPair label="Order #" value={detail.orderNumber ?? "—"} />

   662→                <DataPair label="Customer Priority" value={detail.customerPriority ?? "—"} />
   663→                <DataPair label="Order Priority" value={detail.orderPriority ?? "—"} />
   664→                <DataPair label="Requirement Priority" value={detail.requirementPriority ?? "—"} />
   665→                <DataPair label="Priority Reason" value={detail.priorityReason ?? "—"} />
   666→              </div>
   667→
   668→              {/* Allocations */}
   669→              <div className="rounded-md border border-border overflow-hidden">
   670→                <div className="px-2 py-1.5 border-b border-border bg-muted/40 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   671→                  Plan Allocations ({detail.allocations.length})
   672→                </div>
   673→                {detail.allocations.length === 0 ? (

   675→                ) : (

   684→                      </tr>

   687→                      {detail.allocations.map((a) => (

   689→                          <td className="px-2 py-1 font-medium">{a.planOptionCode ?? "—"}</td>

   692→                          <td className="px-2 py-1">{fmtDate(a.allocatedAt)}</td>
   693→                          <td className="px-2 py-1"><StatusBadge status={a.status} /></td>
   694→                        </tr>
   695→                      ))}
   696→                    </tbody>
   697→                  </table>
   698→                )}
   699→              </div>
   700→
   701→              {/* Source records */}
   702→              <div className="rounded-md border border-border overflow-hidden">
   703→                <div className="px-2 py-1.5 border-b border-border bg-muted/40 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   704→                  Source Records (JSON)
   705→                </div>
   706→                <pre className="text-[10px] leading-relaxed p-2 overflow-x-auto max-h-48 bg-muted/20">
   707→                  {JSON.stringify(detail.sourceRecords, null, 2)}
   708→                </pre>
   709→              </div>
   710→
   711→              <InfoBanner variant="info">
   712→                Four numbers are computed via deterministic formulas with decimal-safe round-half-up. Each step is auditable.
   713→              </InfoBanner>
   714→            </div>
   715→          )}
   716→        </DialogContent>
   717→      </Dialog>
   718→    </div>
   719→  );
   720→}
   721→
   722→function DataPair({ label, value }: { label: string; value: string | number }) {
   723→  return (
   724→    <div className="flex flex-col">

   726→      <span className="text-[11px] font-medium text-foreground tabular-nums">{value}</span>
   727→    </div>
   728→  );
   729→}
