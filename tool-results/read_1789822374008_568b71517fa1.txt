   185→function WarningsCell({ value }: { value: string | null }) {
   186→  if (!value) return <span className="text-muted-foreground">—</span>;
   187→  let parsed: string[] = [];
   188→  try {
   189→    const j = JSON.parse(value);
   190→    if (Array.isArray(j)) parsed = j.map((s) => String(s));
   191→    else if (typeof j === "string") parsed = [j];
   192→    else parsed = [JSON.stringify(j)];
   193→  } catch {
   194→    parsed = [value];
   195→  }
   196→  if (parsed.length === 0) return <span className="text-muted-foreground">—</span>;
   197→  return (
   198→    <div className="flex flex-wrap gap-1">
   199→      {parsed.slice(0, 3).map((w, i) => (
   200→        <Badge key={i} variant="warning">{w}</Badge>
   201→      ))}
   202→      {parsed.length > 3 && (
   203→        <Badge variant="neutral">+{parsed.length - 3}</Badge>
   204→      )}
   205→    </div>
   206→  );
   207→}
