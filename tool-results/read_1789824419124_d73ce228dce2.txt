   153→function yieldIntent(y: number): "default" | "warning" | "success" | "info" | "critical" {
   154→  if (y >= 12) return "success";
   155→  if (y >= 8) return "info";
   156→  if (y >= 4) return "warning";
   157→  return "critical";
   158→}
   159→
   160→function coverageIntent(c: number): "default" | "warning" | "success" | "info" | "critical" {
   161→  if (c >= 80) return "success";
   162→  if (c >= 50) return "info";
   163→  if (c >= 25) return "warning";
   164→  return "critical";
   165→}
   166→
   167→function parseWarnings(raw: string | null): string[] {
   168→  if (!raw) return [];
   169→  try {
   170→    const j = JSON.parse(raw);
   171→    if (Array.isArray(j)) return j.map((s) => String(s));
   172→    if (typeof j === "string") return [j];
   173→    return [JSON.stringify(j)];
   174→  } catch {
   175→    return [raw];
   176→  }
   177→}

   182→function WarningsCell({ value }: { value: string | null }) {
   183→  const parsed = parseWarnings(value);
   184→  if (parsed.length === 0) return <span className="text-muted-foreground/50">—</span>;
   185→  return (
   186→    <div className="flex flex-wrap gap-1">
   187→      {parsed.slice(0, 2).map((w, i) => (
   188→        <Badge key={i} variant="warning">{w}</Badge>
   189→      ))}
   190→      {parsed.length > 2 && (
   191→        <Badge variant="neutral">+{parsed.length - 2}</Badge>
   192→      )}
   193→    </div>
   194→  );
   195→}
