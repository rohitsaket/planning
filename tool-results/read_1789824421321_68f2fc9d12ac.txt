    98→function fmtSigned(v: number): string {
    99→  if (!Number.isFinite(v)) return "—";
   100→  const s = v > 0 ? "+" : "";
   101→  return `${s}${v.toFixed(2)}`;
   102→}

   117→    const slice = historical.map((h) => h.actualYield);

   120→  }, [historical, summary]);

   122→    // MA5 trailing window — visualise the actual series the MA was built on
   123→    const slice = historical.slice(-5).map((h) => h.actualYield);
   124→    while (slice.length < 2) slice.push(summary?.movingAverageYield ?? 0);
   125→    return slice;
   126→  }, [historical, summary]);
   127→  const naiveSpark = useMemo(() => {
   128→    const slice = historical.slice(-3).map((h) => h.actualYield);
   129→    while (slice.length < 2) slice.push(summary?.naiveLastPeriodYield ?? 0);
   130→    return slice;
   131→  }, [historical, summary]);
   132→  const expSpark = useMemo(() => {
   133→    if (historical.length < 2) {
   134→      return [summary?.exponentialSmoothedYield ?? 0, summary?.exponentialSmoothedYield ?? 0];
   135→    }
   136→    // Recursive ES trail α=0.3
   137→    const trail: number[] = [historical[0].actualYield];
   138→    let s = historical[0].actualYield;
   139→    for (let i = 1; i < historical.length; i++) {
   140→      s = 0.3 * historical[i].actualYield + 0.7 * s;
   141→      trail.push(s);
   142→    }
   143→    return trail;
   144→  }, [historical]);
   145→  const stdSpark = useMemo(() => {
   146→    if (historical.length < 2) {
   147→      return [summary?.stdDev ?? 0, summary?.stdDev ?? 0];
   148→    }
   149→    // Rolling std-dev trail (population)
   150→    const trail: number[] = [];
   151→    for (let i = 0; i < historical.length; i++) {
   152→      const slice = historical.slice(0, i + 1).map((h) => h.actualYield);
   153→      const m = slice.reduce((a, b) => a + b, 0) / slice.length;
   154→      const v = slice.reduce((a, b) => a + (b - m) ** 2, 0) / slice.length;
   155→      trail.push(Math.sqrt(v));
   156→    }
   157→    return trail;
   158→  }, [historical]);

   160→    const slice = historical.map((h) => Math.abs(h.variance));
