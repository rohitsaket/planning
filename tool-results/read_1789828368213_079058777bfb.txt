   229→function GlobalSearch() {
   230→  const [query, setQuery] = useState("");
   231→  const [open, setOpen] = useState(false);
   232→  const setView = useNavStore((s) => s.setView);
   233→
   234→  const { data: searchResults } = useQuery({
   235→    queryKey: ["global-search", query],
   236→    queryFn: async () => {
   237→      if (!query || query.length < 2) return null;
   238→      // Search across rough, polished, requirements, cases
   239→      try {
   240→        const [roughRes, polishedRes, reqRes] = await Promise.all([
   241→          apiFetch<{ rows: Array<{ id: string; fantasyRoughId: string; stoneName: string; kapan: string }> }>(`/api/planning/rough?q=${encodeURIComponent(query)}`).catch(() => null),
   242→          apiFetch<{ rows: Array<{ id: string; fantasyLotId: string; shape: string; weight: number }> }>(`/api/fantasy/polished?q=${encodeURIComponent(query)}`).catch(() => null),
   243→          apiFetch<{ data: Array<{ id: string; requirementCode: string; customerName: string | null }> }>(`/api/requirements?q=${encodeURIComponent(query)}`).catch(() => null),
   244→        ]);
   245→        return { rough: roughRes?.rows?.slice(0, 5) ?? [], polished: polishedRes?.rows?.slice(0, 5) ?? [], requirements: reqRes?.data?.slice(0, 5) ?? [] };
   246→      } catch {
   247→        return null;
   248→      }
   249→    },
   250→    enabled: query.length >= 2,
   251→    staleTime: 10_000,
   252→  });
   253→
   254→  return (
   255→    <div className="relative w-full max-w-md">
   256→      <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   257→      <Input
   258→        value={query}
   259→        onChange={(e) => { setQuery(e.target.value); setOpen(true); }}
   260→        onFocus={() => setOpen(true)}
   261→        onBlur={() => setTimeout(() => setOpen(false), 200)}
   262→        placeholder="Search Lot ID, Rough ID, Kapan, Requirement..."
   263→        className="h-8 pl-8 text-xs bg-muted/50 border-border/60"
   264→      />
   265→      {open && searchResults && (searchResults.rough.length > 0 || searchResults.polished.length > 0 || searchResults.requirements.length > 0) && (
   266→        <div className="absolute top-full mt-1 left-0 right-0 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   267→          {searchResults.rough.length > 0 && (
   268→            <div className="p-2">
   269→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Rough Stones</p>
   270→              {searchResults.rough.map((r) => (
   271→                <button key={r.id} onClick={() => { setView("fantasy-rough"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   272→                  <Gem className="h-3 w-3 text-muted-foreground" />
   273→                  <span className="font-medium">{r.fantasyRoughId}</span>
   274→                  <span className="text-muted-foreground truncate">{r.stoneName}</span>
   275→                </button>
   276→              ))}
   277→            </div>
   278→          )}
   279→          {searchResults.polished.length > 0 && (
   280→            <div className="p-2 border-t border-border">
   281→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Polished Lots</p>
   282→              {searchResults.polished.map((p) => (
   283→                <button key={p.id} onClick={() => { setView("fantasy-polished"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   284→                  <Diamond className="h-3 w-3 text-muted-foreground" />
   285→                  <span className="font-medium">{p.fantasyLotId}</span>
   286→                  <span className="text-muted-foreground">{p.shape} {p.weight}ct</span>
   287→                </button>
   288→              ))}
   289→            </div>
   290→          )}
   291→          {searchResults.requirements.length > 0 && (
   292→            <div className="p-2 border-t border-border">
   293→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Requirements</p>
   294→              {searchResults.requirements.map((r) => (
   295→                <button key={r.id} onClick={() => { setView("requirements-matrix"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   296→                  <ClipboardList className="h-3 w-3 text-muted-foreground" />
   297→                  <span className="font-medium">{r.requirementCode}</span>
   298→                  <span className="text-muted-foreground truncate">{r.customerName ?? ""}</span>
   299→                </button>
   300→              ))}
   301→            </div>
   302→          )}
   303→        </div>
   304→      )}
   305→    </div>
   306→  );
   307→}

   309→function ThemeToggle() {
   310→  const { theme, setTheme } = useTheme();
   311→  const [mounted, setMounted] = useState(false);
   312→  // eslint-disable-next-line react-hooks/set-state-in-effect
   313→  useEffect(() => setMounted(true), []);
   314→  if (!mounted) return <div className="h-8 w-8" />;
   315→  return (
   316→    <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setTheme(theme === "dark" ? "light" : "dark")}>
   317→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   318→    </Button>
   319→  );
   320→}

   322→function NotificationsBell() {
   323→  const { data } = useQuery({
   324→    queryKey: ["notifications"],
   325→    queryFn: () => apiFetch<{ rows: Array<{ id: string; title: string; message: string; severity: string; read: boolean }> }>("/api/notifications"),
   326→    staleTime: 60_000,
   327→  });
   328→  const unread = data?.rows.filter((n) => !n.read).length ?? 0;
   329→  const [open, setOpen] = useState(false);
   330→  // Realtime notifications
   331→  const realtimeEvents = useRealtimeStore((s) => s.events);
   332→  const realtimeConnected = useRealtimeStore((s) => s.connected);
   333→  const realtimeUnread = useRealtimeStore((s) => s.unreadCount);
   334→  const clearRealtimeUnread = useRealtimeStore((s) => s.clearUnread);
   335→  const totalUnread = unread + realtimeUnread;
   336→
   337→  const handleToggle = () => {
   338→    const next = !open;
   339→    setOpen(next);
   340→    if (next && realtimeUnread > 0) clearRealtimeUnread();
   341→  };
   342→
   343→  const formatRelTime = (iso: string) => {
   344→    const diff = Date.now() - new Date(iso).getTime();
   345→    const sec = Math.floor(diff / 1000);
   346→    if (sec < 60) return `${sec}s ago`;
   347→    const min = Math.floor(sec / 60);
   348→    if (min < 60) return `${min}m ago`;
   349→    const hr = Math.floor(min / 60);
   350→    return `${hr}h ago`;
   351→  };
   352→
   353→  return (
   354→    <div className="relative">
   355→      <Button variant="ghost" size="icon" className="h-8 w-8 relative" onClick={handleToggle}>
   356→        <Bell className="h-4 w-4" />
   357→        {/* Live indicator when connected */}
   358→        {realtimeConnected && (
   359→          <span className="absolute top-1 right-1 h-2 w-2 rounded-full bg-emerald-500 ring-2 ring-card animate-pulse" title="Live — connected to realtime service" />
   360→        )}
   361→        {totalUnread > 0 && (
   362→          <span className="absolute -top-0.5 -right-0.5 h-3.5 min-w-3.5 px-1 rounded-full bg-rose-500 text-white text-[9px] font-bold flex items-center justify-center">
   363→            {totalUnread > 9 ? "9+" : totalUnread}
   364→          </span>
   365→        )}
   366→      </Button>
   367→      {open && (
   368→        <div className="absolute top-full mt-1 right-0 w-80 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   369→          <div className="px-3 py-2 border-b border-border bg-muted/50 flex items-center justify-between">
   370→            <p className="text-xs font-semibold">Notifications</p>
   371→            <span className={cn("flex items-center gap-1 text-[10px] font-medium px-1.5 py-0.5 rounded",
   372→              realtimeConnected ? "text-emerald-700 bg-emerald-100 dark:text-emerald-300 dark:bg-emerald-950/40" : "text-muted-foreground bg-muted")}>
   373→              <span className={cn("h-1.5 w-1.5 rounded-full", realtimeConnected ? "bg-emerald-500" : "bg-muted-foreground")} />
   374→              {realtimeConnected ? "Live" : "Offline"}
   375→            </span>
   376→          </div>
   377→          <div className="max-h-80 overflow-y-auto">
   378→            {/* Realtime events first */}
   379→            {realtimeEvents.length > 0 && (
   380→              <>
   381→                <div className="px-3 py-1.5 bg-sky-50/50 dark:bg-sky-950/20 border-b border-border/50">
   382→                  <p className="text-[10px] font-semibold uppercase tracking-wide text-sky-700 dark:text-sky-300">Realtime ({realtimeEvents.length})</p>
   383→                </div>
