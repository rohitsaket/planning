    99→function fmtSigned(v: number): string {
   100→  if (!Number.isFinite(v)) return "—";
   101→  const s = v > 0 ? "+" : "";
   102→  return `${s}${v.toFixed(2)}`;
   103→}

   135→      return [summary?.exponentialSmoothedYield ?? 0, summary?.exponentialSmoothedYield ?? 0];

   140→    for (let i = 1; i < historical.length; i++) {
   141→      s = 0.3 * historical[i].actualYield + 0.7 * s;
   142→      trail.push(s);
   143→    }

   148→      return [summary?.stdDev ?? 0, summary?.stdDev ?? 0];

   155→      const v = slice.reduce((a, b) => a + (b - m) ** 2, 0) / slice.length;
