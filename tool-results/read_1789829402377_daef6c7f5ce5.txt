   105→function fmtRunDate(iso: string | null): string {
   106→  if (!iso) return "—";
   107→  const d = new Date(iso);
   108→  if (Number.isNaN(d.getTime())) return "—";
   109→  return d.toISOString().slice(0, 16).replace("T", " ");
   110→}
   111→
   112→function formatNum(v: number): string {
   113→  if (!Number.isFinite(v)) return "0";
   114→  if (Number.isInteger(v)) return v.toString();
   115→  return v.toFixed(2);
   116→}

   182→// ---------------------------------------------------------------------------
   183→// Four Requirement Numbers card

   185→function FourNumberCard({
   186→  label, value, formula, icon: Icon, intent,
   187→}: {
   188→  label: string;
   189→  value: number;
   190→  formula: string;
   191→  icon: LucideIcon;
   192→  intent: "critical" | "warning" | "info" | "success";
   193→}) {
   194→  const styles: Record<string, { ring: string; text: string; chip: string }> = {
   195→    critical: { ring: "border-rose-300 dark:border-rose-900", text: "text-rose-700 dark:text-rose-300", chip: "bg-rose-500/15 text-rose-600 dark:text-rose-400" },
   196→    warning: { ring: "border-amber-300 dark:border-amber-900", text: "text-amber-700 dark:text-amber-300", chip: "bg-amber-500/15 text-amber-600 dark:text-amber-400" },
   197→    info: { ring: "border-sky-300 dark:border-sky-900", text: "text-sky-700 dark:text-sky-300", chip: "bg-sky-500/15 text-sky-600 dark:text-sky-400" },
   198→    success: { ring: "border-emerald-300 dark:border-emerald-900", text: "text-emerald-700 dark:text-emerald-300", chip: "bg-emerald-500/15 text-emerald-600 dark:text-emerald-400" },
   199→  };
   200→  const s = styles[intent];
   201→  return (
   202→    <div className={cn("rounded-md border p-3 bg-card flex flex-col gap-1.5", s.ring)}>
   203→      <div className="flex items-center justify-between gap-2">
   204→        <span className="text-[10px] uppercase tracking-wide text-muted-foreground font-semibold">
   205→          {label}
   206→        </span>
   207→        <span className={cn("p-1 rounded-md", s.chip)}>
   208→          <Icon className="h-3 w-3" />
   209→        </span>
   210→      </div>
   211→      <div className={cn("text-3xl font-bold tabular-nums leading-none", s.text)}>
   212→        {value}
   213→      </div>
   214→      <div className="text-[10px] font-mono text-muted-foreground leading-snug">
   215→        {formula}
   216→      </div>
   217→    </div>
   218→  );
   219→}
