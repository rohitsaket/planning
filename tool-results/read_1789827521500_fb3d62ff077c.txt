   228→function GlobalSearch() {
   229→  const [query, setQuery] = useState("");
   230→  const [open, setOpen] = useState(false);
   231→  const setView = useNavStore((s) => s.setView);
   232→
   233→  const { data: searchResults } = useQuery({
   234→    queryKey: ["global-search", query],
   235→    queryFn: async () => {
   236→      if (!query || query.length < 2) return null;
   237→      // Search across rough, polished, requirements, cases
   238→      try {
   239→        const [roughRes, polishedRes, reqRes] = await Promise.all([
   240→          apiFetch<{ rows: Array<{ id: string; fantasyRoughId: string; stoneName: string; kapan: string }> }>(`/api/planning/rough?q=${encodeURIComponent(query)}`).catch(() => null),
   241→          apiFetch<{ rows: Array<{ id: string; fantasyLotId: string; shape: string; weight: number }> }>(`/api/fantasy/polished?q=${encodeURIComponent(query)}`).catch(() => null),
   242→          apiFetch<{ data: Array<{ id: string; requirementCode: string; customerName: string | null }> }>(`/api/requirements?q=${encodeURIComponent(query)}`).catch(() => null),
   243→        ]);
   244→        return { rough: roughRes?.rows?.slice(0, 5) ?? [], polished: polishedRes?.rows?.slice(0, 5) ?? [], requirements: reqRes?.data?.slice(0, 5) ?? [] };
   245→      } catch {
   246→        return null;
   247→      }
   248→    },
   249→    enabled: query.length >= 2,
   250→    staleTime: 10_000,
   251→  });
   252→
   253→  return (
   254→    <div className="relative w-full max-w-md">
   255→      <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   256→      <Input
   257→        value={query}
   258→        onChange={(e) => { setQuery(e.target.value); setOpen(true); }}
   259→        onFocus={() => setOpen(true)}
   260→        onBlur={() => setTimeout(() => setOpen(false), 200)}
   261→        placeholder="Search Lot ID, Rough ID, Kapan, Requirement..."
   262→        className="h-8 pl-8 text-xs bg-muted/50 border-border/60"
   263→      />
   264→      {open && searchResults && (searchResults.rough.length > 0 || searchResults.polished.length > 0 || searchResults.requirements.length > 0) && (
   265→        <div className="absolute top-full mt-1 left-0 right-0 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   266→          {searchResults.rough.length > 0 && (
   267→            <div className="p-2">
   268→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Rough Stones</p>
   269→              {searchResults.rough.map((r) => (
   270→                <button key={r.id} onClick={() => { setView("fantasy-rough"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   271→                  <Gem className="h-3 w-3 text-muted-foreground" />
   272→                  <span className="font-medium">{r.fantasyRoughId}</span>
   273→                  <span className="text-muted-foreground truncate">{r.stoneName}</span>
   274→                </button>
   275→              ))}
   276→            </div>
   277→          )}
   278→          {searchResults.polished.length > 0 && (
   279→            <div className="p-2 border-t border-border">
   280→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Polished Lots</p>
   281→              {searchResults.polished.map((p) => (
   282→                <button key={p.id} onClick={() => { setView("fantasy-polished"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   283→                  <Diamond className="h-3 w-3 text-muted-foreground" />
   284→                  <span className="font-medium">{p.fantasyLotId}</span>
   285→                  <span className="text-muted-foreground">{p.shape} {p.weight}ct</span>
   286→                </button>
   287→              ))}
   288→            </div>
   289→          )}
   290→          {searchResults.requirements.length > 0 && (
   291→            <div className="p-2 border-t border-border">
   292→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground px-1 mb-1">Requirements</p>
   293→              {searchResults.requirements.map((r) => (
   294→                <button key={r.id} onClick={() => { setView("requirements-matrix"); setOpen(false); }} className="w-full flex items-center gap-2 px-2 py-1 text-xs hover:bg-muted rounded text-left">
   295→                  <ClipboardList className="h-3 w-3 text-muted-foreground" />
   296→                  <span className="font-medium">{r.requirementCode}</span>
   297→                  <span className="text-muted-foreground truncate">{r.customerName ?? ""}</span>
   298→                </button>
   299→              ))}
   300→            </div>
   301→          )}
   302→        </div>
   303→      )}
   304→    </div>
   305→  );
   306→}

   308→function ThemeToggle() {
   309→  const { theme, setTheme } = useTheme();
   310→  const [mounted, setMounted] = useState(false);
   311→  // eslint-disable-next-line react-hooks/set-state-in-effect
   312→  useEffect(() => setMounted(true), []);
   313→  if (!mounted) return <div className="h-8 w-8" />;
   314→  return (
   315→    <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setTheme(theme === "dark" ? "light" : "dark")}>
   316→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   317→    </Button>
   318→  );
   319→}

   321→function NotificationsBell() {
   322→  const { data } = useQuery({
   323→    queryKey: ["notifications"],
   324→    queryFn: () => apiFetch<{ rows: Array<{ id: string; title: string; message: string; severity: string; read: boolean }> }>("/api/notifications"),
   325→    staleTime: 60_000,
   326→  });
   327→  const unread = data?.rows.filter((n) => !n.read).length ?? 0;
   328→  const [open, setOpen] = useState(false);
   329→  // Realtime notifications
   330→  const realtimeEvents = useRealtimeStore((s) => s.events);
   331→  const realtimeConnected = useRealtimeStore((s) => s.connected);
   332→  const realtimeUnread = useRealtimeStore((s) => s.unreadCount);
   333→  const clearRealtimeUnread = useRealtimeStore((s) => s.clearUnread);
   334→  const totalUnread = unread + realtimeUnread;
   335→
   336→  const handleToggle = () => {
   337→    const next = !open;
   338→    setOpen(next);
   339→    if (next && realtimeUnread > 0) clearRealtimeUnread();
   340→  };
   341→
   342→  const formatRelTime = (iso: string) => {
   343→    const diff = Date.now() - new Date(iso).getTime();
   344→    const sec = Math.floor(diff / 1000);
   345→    if (sec < 60) return `${sec}s ago`;
   346→    const min = Math.floor(sec / 60);
   347→    if (min < 60) return `${min}m ago`;
   348→    const hr = Math.floor(min / 60);
   349→    return `${hr}h ago`;
   350→  };
   351→
   352→  return (
   353→    <div className="relative">
   354→      <Button variant="ghost" size="icon" className="h-8 w-8 relative" onClick={handleToggle}>
   355→        <Bell className="h-4 w-4" />
   356→        {/* Live indicator when connected */}
   357→        {realtimeConnected && (
   358→          <span className="absolute top-1 right-1 h-2 w-2 rounded-full bg-emerald-500 ring-2 ring-card animate-pulse" title="Live — connected to realtime service" />
   359→        )}
   360→        {totalUnread > 0 && (
   361→          <span className="absolute -top-0.5 -right-0.5 h-3.5 min-w-3.5 px-1 rounded-full bg-rose-500 text-white text-[9px] font-bold flex items-center justify-center">
   362→            {totalUnread > 9 ? "9+" : totalUnread}
   363→          </span>
   364→        )}
   365→      </Button>
   366→      {open && (
   367→        <div className="absolute top-full mt-1 right-0 w-80 z-50 rounded-md border border-border bg-popover shadow-lg overflow-hidden">
   368→          <div className="px-3 py-2 border-b border-border bg-muted/50 flex items-center justify-between">
   369→            <p className="text-xs font-semibold">Notifications</p>
   370→            <span className={cn("flex items-center gap-1 text-[10px] font-medium px-1.5 py-0.5 rounded",
   371→              realtimeConnected ? "text-emerald-700 bg-emerald-100 dark:text-emerald-300 dark:bg-emerald-950/40" : "text-muted-foreground bg-muted")}>
   372→              <span className={cn("h-1.5 w-1.5 rounded-full", realtimeConnected ? "bg-emerald-500" : "bg-muted-foreground")} />
   373→              {realtimeConnected ? "Live" : "Offline"}
   374→            </span>
   375→          </div>
   376→          <div className="max-h-80 overflow-y-auto">
   377→            {/* Realtime events first */}
   378→            {realtimeEvents.length > 0 && (
   379→              <>
   380→                <div className="px-3 py-1.5 bg-sky-50/50 dark:bg-sky-950/20 border-b border-border/50">
   381→                  <p className="text-[10px] font-semibold uppercase tracking-wide text-sky-700 dark:text-sky-300">Realtime ({realtimeEvents.length})</p>
   382→                </div>
   383→                {realtimeEvents.map((evt) => (
