   154→function yieldIntent(y: number): "default" | "warning" | "success" | "info" | "critical" {
   155→  if (y >= 12) return "success";
   156→  if (y >= 8) return "info";
   157→  if (y >= 4) return "warning";
   158→  return "critical";
   159→}
   160→
   161→function coverageIntent(c: number): "default" | "warning" | "success" | "info" | "critical" {
   162→  if (c >= 80) return "success";
   163→  if (c >= 50) return "info";
   164→  if (c >= 25) return "warning";
   165→  return "critical";
   166→}
   167→
   168→function parseWarnings(raw: string | null): string[] {
   169→  if (!raw) return [];
   170→  try {
   171→    const j = JSON.parse(raw);
   172→    if (Array.isArray(j)) return j.map((s) => String(s));
   173→    if (typeof j === "string") return [j];
   174→    return [JSON.stringify(j)];
   175→  } catch {
   176→    return [raw];
   177→  }
   178→}

   183→function WarningsCell({ value }: { value: string | null }) {
   184→  const parsed = parseWarnings(value);
   185→  if (parsed.length === 0) return <span className="text-muted-foreground/50">—</span>;
   186→  return (
   187→    <div className="flex flex-wrap gap-1">
   188→      {parsed.slice(0, 2).map((w, i) => (
   189→        <Badge key={i} variant="warning">{w}</Badge>
   190→      ))}
   191→      {parsed.length > 2 && (
   192→        <Badge variant="neutral">+{parsed.length - 2}</Badge>
   193→      )}
   194→    </div>
   195→  );
   196→}
