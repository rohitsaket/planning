   130→   130→
   131→   131→function priorityVariant(
   132→   132→  v: string | null
   133→   133→): "default" | "critical" | "high" | "low" | "neutral" {
   134→   134→  if (!v) return "default";
   135→   135→  if (v === "CRITICAL") return "critical";
   136→   136→  if (v === "HIGH") return "high";
   137→   137→  if (v === "LOW") return "low";
   138→   138→  if (v === "WATCH") return "neutral";
   139→   139→  return "default";
   140→   140→}
   141→   141→
   142→   142→function PriorityBadge({ value }: { value: string | null }) {
   143→   143→  if (!value) return <span className="text-muted-foreground">—</span>;
   144→   144→  return <Badge variant={priorityVariant(value)}>{value}</Badge>;
   145→   145→}

   147→   147→function fmtDate(iso: string | null): string {
   148→   148→  if (!iso) return "—";

   151→   151→  } catch {
